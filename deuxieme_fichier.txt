          <div class="pill">Meilleur: <span id="bestScore">0</span></div>

      <canvas id="game" width="560" height="560" aria-label="Jeu Snake"></canvas>

      <div class="controls">
        <button id="restartBtn" type="button">Rejouer</button>
        <div class="help">Touches : ↑ ↓ ← → ou WASD</div>
      </div>
    </div>

    <script>
      const canvas = document.getElementById('game');
      const ctx = canvas.getContext('2d');
      const scoreEl = document.getElementById('score');
      const bestScoreEl = document.getElementById('bestScore');
      const restartBtn = document.getElementById('restartBtn');

      const gridSize = 20;
      const cellSize = canvas.width / gridSize;
      const baseSpeed = 120;

      let snake;
      let direction;
      let nextDirection;
      let food;
      let score;
      let bestScore = Number(localStorage.getItem('snakeBestScore')) || 0;
      let loopId;
      let tickSpeed;

      function updateBestScore() {
        bestScoreEl.textContent = bestScore;
      }

      function resetGame() {
        snake = [
          { x: 10, y: 10 },
          { x: 9, y: 10 },
          { x: 8, y: 10 }
        ];
        direction = { x: 1, y: 0 };
        nextDirection = { x: 1, y: 0 };
        score = 0;
        tickSpeed = baseSpeed;
        placeFood();
        scoreEl.textContent = score;
        updateBestScore();
        clearInterval(loopId);
        loopId = setInterval(gameLoop, tickSpeed);
      }

      function placeFood() {
        let valid = true;
        do {
          valid = true;
          food = {
            x: Math.floor(Math.random() * gridSize),
            y: Math.floor(Math.random() * gridSize)
          };

          for (const part of snake) {
            if (part.x === food.x && part.y === food.y) {
              valid = false;
              break;
            }
          }
        } while (!valid);
      }

      function drawCell(x, y, color) {
        ctx.fillStyle = color;
        ctx.fillRect(x * cellSize + 1, y * cellSize + 1, cellSize - 2, cellSize - 2);
      }

      function drawGrid() {
        ctx.fillStyle = '#0b1220';
        ctx.fillRect(0, 0, canvas.width, canvas.height);

        ctx.strokeStyle = 'rgba(148, 163, 184, 0.12)';
        ctx.lineWidth = 1;

        for (let i = 0; i <= gridSize; i++) {
          const pos = i * cellSize;
          ctx.beginPath();
          ctx.moveTo(pos, 0);
          ctx.lineTo(pos, canvas.height);
          ctx.stroke();

          ctx.beginPath();
          ctx.moveTo(0, pos);
          ctx.lineTo(canvas.width, pos);
          ctx.stroke();
        }
      }

      function drawFood() {
        drawCell(food.x, food.y, '#ef4444');
      }

      function drawSnake() {
        snake.forEach((part, index) => {
          drawCell(part.x, part.y, index === 0 ? '#22c55e' : '#16a34a');
        });
      }

      function updateDirection(newDir) {
        const isOpposite = newDir.x === -direction.x && newDir.y === -direction.y;
        if (!isOpposite) {
          nextDirection = newDir;
        }
      }

      function gameLoop() {
        direction = nextDirection;
        const head = { x: snake[0].x + direction.x, y: snake[0].y + direction.y };

        const hitWall = head.x < 0 || head.y < 0 || head.x >= gridSize || head.y >= gridSize;
        const hitSelf = snake.some((part) => part.x === head.x && part.y === head.y);

        if (hitWall || hitSelf) {
          if (score > bestScore) {
            bestScore = score;
            localStorage.setItem('snakeBestScore', String(bestScore));
            updateBestScore();
          }
          resetGame();
          return;
        }

        snake.unshift(head);

        if (head.x === food.x && head.y === food.y) {
          score += 10;
          scoreEl.textContent = score;
          tickSpeed = Math.max(55, baseSpeed - score * 0.7);
          clearInterval(loopId);
          loopId = setInterval(gameLoop, tickSpeed);
          placeFood();
        } else {
          snake.pop();
        }

        drawGrid();
        drawFood();
        drawSnake();
      }

      document.addEventListener('keydown', (event) => {
        const key = event.key.toLowerCase();

        if (key === 'arrowup' || key === 'z') updateDirection({ x: 0, y: -1 });
        if (key === 'arrowdown' || key === 's') updateDirection({ x: 0, y: 1 });
        if (key === 'arrowleft' || key === 'q') updateDirection({ x: -1, y: 0 });
        if (key === 'arrowright' || key === 'd') updateDirection({ x: 1, y: 0 });
      });

      restartBtn.addEventListener('click', resetGame);

      resetGame();
      drawGrid();
      drawFood();
      drawSnake();
    </script>
